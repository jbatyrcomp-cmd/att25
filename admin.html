⚡ Supabase SQL Jadvallar Sxemasi

Agar Supabase loyihangizda categories va materials jadvallari hali ochilmagan bo'lsa, quyidagi SQL kodini nusxalang va Supabase panelidagi SQL Editor bo'limida Run tugmasini bosing: 

-- 1. Bo'limlar (categories) jadvali
CREATE TABLE IF NOT EXISTS categories (
  id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  name TEXT NOT NULL UNIQUE,
  created_at TIMESTAMPTZ DEFAULT NOW()
);

-- 2. Materiallar (materials) jadvali
CREATE TABLE IF NOT EXISTS materials (
  id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  category_id BIGINT REFERENCES categories(id) ON DELETE CASCADE,
  title TEXT NOT NULL,
  type TEXT NOT NULL DEFAULT 'other',
  url TEXT NOT NULL,
  created_at TIMESTAMPTZ DEFAULT NOW()
);

-- 3. RLS (Row Level Security) xavfsizlik qoidalari:
ALTER TABLE categories ENABLE ROW LEVEL SECURITY;
ALTER TABLE materials ENABLE ROW LEVEL SECURITY;

-- Saytga kirgan har kim materiallarni ko'ra oladi (SELECT)
CREATE POLICY "Public categories read" ON categories FOR SELECT USING (true);
CREATE POLICY "Public materials read" ON materials FOR SELECT USING (true);

-- Tizimga kirgan adminlar to'liq qo'shish, o'zgartirish va o'chirish huquqiga ega
CREATE POLICY "Admin categories insert" ON categories FOR INSERT TO authenticated WITH CHECK (true);
CREATE POLICY "Admin categories update" ON categories FOR UPDATE TO authenticated USING (true);
CREATE POLICY "Admin categories delete" ON categories FOR DELETE TO authenticated USING (true);

CREATE POLICY "Admin materials insert" ON materials FOR INSERT TO authenticated WITH CHECK (true);
CREATE POLICY "Admin materials update" ON materials FOR UPDATE TO authenticated USING (true);
CREATE POLICY "Admin materials delete" ON materials FOR DELETE TO authenticated USING (true);
